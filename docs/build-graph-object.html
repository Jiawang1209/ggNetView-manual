  1. Build a graph object
  2. Understand and manipulate the graph object
  3. Visualize the network using layout algorithms

Build graph object

# ---- Build a co-occurrence network directly from an abundance matrix ----
# Internal pipeline: transform -> correlation -> p-value adjustment ->
# edge filtering -> community detection -> attach taxonomy.
graph_obj <- build_graph_from_mat(
  mat              = otu_rare_relative,  # variables x samples numeric matrix
  transfrom.method = "none",             # input is already relative abundance, no extra transform
  r.threshold      = 0.7,                # |r| cutoff (subjective; consider ggNetView_RMT)
  p.threshold      = 0.05,               # adjusted p-value cutoff
  method           = "WGCNA",            # correlation backend: WGCNA::corAndPvalue
  cor.method       = "pearson",          # Pearson correlation
  proc             = "bonferroni",       # multiple-testing correction
  module.method    = "Fast_greedy",      # community detection algorithm
  node_annotation  = tax_tab,            # taxonomy joined onto each node by name
  top_modules      = 15,                 # keep top-15 modules; rest -> "Others"
  seed             = 1115                # fix RNG for reproducibility
)

# Inspect node/edge counts and the columns now attached to nodes:
# modularity, modularity2, modularity3, Modularity, Degree, Strength + taxonomy.

# ---- Build a tbl_graph directly from an edge-list data.frame ----
# No correlation step — edges are taken as-is from `df`.
# Community detection still runs on the resulting graph.
graph_obj_from_df <- build_graph_from_df(

Build graph object

# ---- Cross-domain (bipartite-style) network ----
# Correlations are computed BETWEEN features in mat1 and features in mat2
# (no within-mat1 / within-mat2 edges), then thresholded, modularised,
# and joined with the node-annotation table.
graph_obj_double_mat <- build_graph_from_double_mat(
  mat1            = BASV_tab,            # set A (e.g. bacteria), variables x samples
  mat2            = FASV_tab,            # set B (e.g. fungi),    variables x samples
  module.method   = "Fast_greedy",       # community detection algorithm
  node_annotation = double_mat_node_df,  # node metadata covering BOTH mat1 and mat2
  top_modules     = 15,                  # keep top-15 modules; rest -> "Others"
  seed            = 1115                 # fix RNG for reproducibility
)

  TOM_mat,     # square TOM matrix from WGCNA
  plot.mat,    # original abundance matrix the TOM came from
  top_k = 1    # keep top_k strongest neighbours per node

  ID     = names(net$colors),          # feature IDs
  Module = as.character(net$colors)    # WGCNA module colour as module label
)

id <- data.frame(
  ID = names(sort(net$colors))         # node IDs in module order

  wgcna_tom       = trans_TOM,  # sparsified TOM (from trans_TOM_in_WGCNA)

# ---- Consensus-network workflow ----
# (1) Build several per-method networks from the SAME matrix.
# (2) Convert each tbl_graph back to an adjacency matrix.
# (3) Fuse the per-method adjacencies into ONE consensus tbl_graph.

# Restrict to the 40 most abundant OTUs to keep this chunk responsive.
data(otu_rare_relative)
mat <- as.matrix(otu_rare_relative)
mat <- mat[order(rowSums(mat), decreasing = TRUE)[seq_len(40)], ]

# ---- (1) Three per-method networks on the same data ----

# Spearman correlation through psych::corr.test — baseline method.
g_cor   <- build_graph_from_mat(
  mat,                              # input matrix (variables x samples)
  method        = "cor",            # correlation backend: psych::corr.test
  cor.method    = "spearman",       # Spearman correlation
  proc          = "BH",             # multiple-testing correction (Benjamini-Hochberg)
  r.threshold   = 0.4,              # |r| cutoff
  p.threshold   = 0.05,             # adjusted p-value cutoff
  module.method = "Fast_greedy",    # community detection algorithm
  seed          = 1                 # fix RNG for reproducibility
)
## The max module in network is 5 we use the 5  modules for next analysis
# WGCNA-style correlation (WGCNA::corAndPvalue + the package's thresholding).
g_wgcna <- build_graph_from_mat(
  mat,                              # input matrix (variables x samples)
  method        = "WGCNA",          # correlation backend: WGCNA::corAndPvalue
  cor.method    = "spearman",       # Spearman correlation
  proc          = "BH",             # multiple-testing correction (Benjamini-Hochberg)
  r.threshold   = 0.4,              # |r| cutoff
  p.threshold   = 0.05,             # adjusted p-value cutoff
  module.method = "Fast_greedy",    # community detection algorithm
  seed          = 1                 # fix RNG for reproducibility
)
## The max module in network is 6 we use the 6  modules for next analysis
# Hmisc::rcorr — another correlation engine; slightly different NA/tie behaviour.
g_hmisc <- build_graph_from_mat(
  mat,                              # input matrix (variables x samples)
  method        = "Hmisc",          # correlation backend: Hmisc::rcorr
  cor.method    = "spearman",       # Spearman correlation
  r.threshold   = 0.4,              # |r| cutoff
  p.threshold   = 0.05,             # raw p-value cutoff (no `proc` here)
  module.method = "Fast_greedy",    # community detection algorithm
  seed          = 1                 # fix RNG for reproducibility
)
## The max module in network is 4 we use the 4  modules for next analysis
# ---- (2) tbl_graph -> adjacency matrix ----
# `get_graph_adjacency()` recovers the weighted adjacency matrix from a tbl_graph,
# which is the format expected by build_graph_from_consensus().
adj_cor   <- get_graph_adjacency(g_cor)
adj_wgcna <- get_graph_adjacency(g_wgcna)
adj_hmisc <- get_graph_adjacency(g_hmisc)

# Bundle per-method matrices into a NAMED list.
# Names ("cor", "wgcna", "hmisc") are kept as method labels downstream.
adj_list <- list(cor = adj_cor, wgcna = adj_wgcna, hmisc = adj_hmisc)

# ---- (3) Consensus fusion ----
# Borda rank fusion: ranks every pair by |w| in each method, averages the ranks,
# then min-max rescales to [-1, 1]. Robust to per-method weight-scale differences.
g_borda <- build_graph_from_consensus(
  adj_list              = adj_list,        # named list of per-method adjacency matrices
  method                = "rank_fusion",   # fusion strategy
  rank_fusion_algorithm = "borda",         # specific rank-fusion variant
  threshold             = 0.5,             # final |w| cutoff applied to the fused matrix
  top_modules           = 5,               # keep top-5 modules; rest -> "Others"
  seed                  = 1                # fix RNG for reproducibility
)
